cmake_minimum_required(VERSION 3.22)
project(BteDaw VERSION 0.2.0 LANGUAGES C CXX)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8 /bigobj)
endif()
option(BTE_BUILD_APP "Build the Windows desktop DAW" ${WIN32})
option(BTE_VALIDATE_LINUX "Enable the development-only Linux validation target" OFF)
include(CTest)
if(BUILD_TESTING)
    add_executable(bte_core_tests tests/CoreTests.cpp)
    target_include_directories(bte_core_tests PRIVATE Source)
    add_test(NAME audio_and_assistant COMMAND bte_core_tests)
endif()
if(BTE_BUILD_APP)
    if(NOT WIN32 AND NOT BTE_VALIDATE_LINUX)
        message(FATAL_ERROR "BTE Studio targets Windows x64. Use BTE_VALIDATE_LINUX for development validation only.")
    endif()
    include(FetchContent)
    FetchContent_Declare(JUCE URL https://codeload.github.com/juce-framework/JUCE/tar.gz/7ff644f65d7daa1428c34b9738ebb5c449dc9172
        URL_HASH SHA256=019212d788fcf7cd0a89721face344f5f0cd43e62e6bccd7bbf6096a4c12c73c)
    FetchContent_MakeAvailable(JUCE)
    set(TE_ADD_EXAMPLES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(tracktion URL https://codeload.github.com/Tracktion/tracktion_engine/tar.gz/b0c7dd090a56b27fe01bb394617891906eb34daf
        URL_HASH SHA256=2b1ba6aeb1dd1d3a1a7f900a90690c75c90e4acc943842c2daa3d4d8e9ac7183)
    FetchContent_MakeAvailable(tracktion)
    add_library(BteEngine STATIC Source/ProEngine.cpp)
    target_include_directories(BteEngine PUBLIC Source "${tracktion_SOURCE_DIR}/modules" "${juce_SOURCE_DIR}/modules")
    target_compile_definitions(BteEngine PUBLIC JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_VST=0
        JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_DISPLAY_SPLASH_SCREEN=0
        JUCE_REPORT_APP_USAGE=0 JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1
        TRACKTION_ENABLE_TIMESTRETCH_SOUNDTOUCH=1 TRACKTION_ENABLE_ARA=0
        TRACKTION_ENABLE_REWIRE=0 TRACKTION_ENABLE_CONTROL_SURFACES=0)
    target_compile_definitions(BteEngine PUBLIC "$<$<CONFIG:Debug>:DEBUG=1;_DEBUG=1>")
    target_link_libraries(BteEngine PRIVATE tracktion::tracktion_engine
        tracktion::tracktion_graph tracktion::tracktion_core juce::juce_audio_utils)
    if(UNIX AND NOT APPLE)
        target_link_libraries(BteEngine PUBLIC atomic)
    endif()
    juce_add_gui_app(BteDaw PRODUCT_NAME "BTE Studio" COMPANY_NAME "BteDaw" VERSION ${PROJECT_VERSION})
    target_sources(BteDaw PRIVATE Source/Main.cpp Source/ProStudio.cpp)
    target_link_libraries(BteDaw PRIVATE BteEngine)
    if(BUILD_TESTING)
        juce_add_console_app(bte_pro_tests PRODUCT_NAME "BTE workstation tests")
        target_sources(bte_pro_tests PRIVATE tests/ProTests.cpp)
        target_link_libraries(bte_pro_tests PRIVATE BteEngine)
        juce_add_plugin(BteTestEffect PRODUCT_NAME "BTE Test Effect" COMPANY_NAME "BteDaw"
            PLUGIN_MANUFACTURER_CODE Bted PLUGIN_CODE Btfx FORMATS VST3
            IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
        target_sources(BteTestEffect PRIVATE tests/TestEffect.cpp)
        target_compile_definitions(BteTestEffect PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
        target_link_libraries(BteTestEffect PRIVATE juce::juce_audio_processors)
        add_dependencies(bte_pro_tests BteTestEffect_VST3)
        add_test(NAME workstation COMMAND bte_pro_tests "${CMAKE_CURRENT_BINARY_DIR}/BteTestEffect_artefacts/$<CONFIG>/VST3/BTE Test Effect.vst3")
        set_tests_properties(workstation PROPERTIES TIMEOUT 180)
    endif()
    install(TARGETS BteDaw RUNTIME DESTINATION .)
    install(FILES LICENSE README.md docs/THIRD_PARTY.md DESTINATION .)
    install(DIRECTORY "${juce_SOURCE_DIR}/" DESTINATION licenses/JUCE
        FILES_MATCHING PATTERN "*LICENSE*" PATTERN "*LICENCE*" PATTERN "*COPYING*"
        PATTERN ".git" EXCLUDE)
    install(DIRECTORY "${tracktion_SOURCE_DIR}/modules/" DESTINATION licenses/Tracktion
        FILES_MATCHING PATTERN "*LICENSE*" PATTERN "*LICENCE*" PATTERN "*COPYING*"
        PATTERN "juce" EXCLUDE PATTERN ".git" EXCLUDE)
    install(FILES "${tracktion_SOURCE_DIR}/LICENSE.md" DESTINATION licenses/Tracktion)
    set(CPACK_PACKAGE_NAME "BTE Studio")
    set(CPACK_PACKAGE_VENDOR "BteDaw")
    set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
    if(WIN32)
        set(CPACK_PACKAGE_FILE_NAME "BTE-Studio-${PROJECT_VERSION}-Windows-x64")
    endif()
    set(CPACK_PACKAGE_INSTALL_DIRECTORY "BTE Studio")
    set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
    set(CPACK_NSIS_DISPLAY_NAME "BTE Studio")
    set(CPACK_NSIS_PACKAGE_NAME "BTE Studio")
    set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
    set(CPACK_PACKAGE_EXECUTABLES "BTE Studio" "BTE Studio")
    set(CPACK_NSIS_EXECUTABLES_DIRECTORY ".")
    set(CPACK_NSIS_INSTALL_ROOT "$PROGRAMFILES64")
    include(CPack)
endif()
